# Wake-phrase benchmark corpus for bin/wake_accuracy.dart.
#
# Same convention as commands.txt: a leading "!" marks a line that must NOT
# trigger. Positives are the wake phrase alone or followed by a command.
#
# Pronunciation variety is produced by spelling: SAPI says "wolsey", "walsy"
# and "wallsey" differently from "walsey", which exercises the matcher's
# canonical forms. This is still synthetic speech; human accents and room noise
# remain unmeasured until recorded audio exists.
#
# Generate with:
#   powershell -File tools/voice_accuracy/make_tts_corpus.ps1 `
#     -CommandFile tools/voice_accuracy/wake.txt `
#     -OutputDirectory %TEMP%\walsey_wake_corpus
# Then:
#   dart run bin/wake_accuracy.dart --model <vosk model> --corpus <that folder>
